### 1. Scene 1: Hero Vortex Funnel (`src/three/HeroCanvas.jsx`)

Replace or update `src/three/HeroCanvas.jsx` with this high-performance 3D projection canvas that renders the swirling hourglass/funnel vortex with a flared base disc and a central readability vignette in Dark Mode:

```jsx
import React, { useEffect, useRef } from 'react';
import { useTheme } from '../theme/ThemeProvider';

export default function HeroCanvas({ className = '' }) {
  const canvasRef = useRef(null);
  const { theme } = useTheme?.() || { theme: 'dark' };
  const isDark = theme === 'dark' || document.documentElement.classList.contains('dark');

  useEffect(() => {
    if (!isDark) return;
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');

    let animationFrameId;
    let width = (canvas.width = canvas.parentElement.clientWidth || window.innerWidth);
    let height = (canvas.height = canvas.parentElement.clientHeight || window.innerHeight);

    const handleResize = () => {
      if (!canvas.parentElement) return;
      width = canvas.width = canvas.parentElement.clientWidth;
      height = canvas.height = canvas.parentElement.clientHeight;
    };
    window.addEventListener('resize', handleResize);

    // Parametric funnel helper: maps normalized vertical progress t in [0, 1] to radius & Y
    // t = 0 is top wide mouth, t = 0.58 is narrow throat, t = 1.0 is wide bottom floor disc
    const getFunnelProfile = (t) => {
      let radius;
      let y;
      if (t < 0.65) {
        // Upper funnel tapering down into the throat
        const u = t / 0.65; // 0 -> 1
        y = -height * 0.58 + u * (height * 0.82);
        radius = width * 0.045 + Math.pow(1 - u, 2.15) * (width * 0.42);
      } else {
        // Lower throat flaring rapidly into the horizontal floor plane
        const u = (t - 0.65) / 0.35; // 0 -> 1
        y = height * 0.24 + Math.pow(u, 0.7) * (height * 0.24);
        radius = width * 0.045 + Math.pow(u, 1.45) * (width * 0.68);
      }
      return { y, radius };
    };

    // Create flowing particles spiraling along the vortex
    const NUM_PARTICLES = 900;
    const particles = Array.from({ length: NUM_PARTICLES }, () => ({
      t: Math.random(),
      angle: Math.random() * Math.PI * 2,
      speedT: 0.0006 + Math.random() * 0.0012,
      speedAngle: 0.012 + Math.random() * 0.018,
      size: Math.random() < 0.8 ? 1.1 : 1.9,
      alpha: 0.18 + Math.random() * 0.55,
    }));

    let time = 0;

    const render = () => {
      time += 0.012;
      ctx.fillStyle = '#000000';
      ctx.fillRect(0, 0, width, height);

      const cx = width * 0.5;
      const cy = height * 0.46;
      const tilt = 0.28; // Perspective vertical compression of rings

      // 1. Draw dense swirling orbital spline rings
      const NUM_RINGS = 78;
      for (let i = 0; i < NUM_RINGS; i++) {
        const t = ((i / NUM_RINGS) + time * 0.025) % 1;
        const { y, radius } = getFunnelProfile(t);

        // Fade rings near top/bottom edges
        const edgeFade = Math.sin(t * Math.PI);
        const alpha = Math.max(0.03, edgeFade * 0.22);

        ctx.beginPath();
        const segments = 64;
        for (let s = 0; s <= segments; s++) {
          const theta = (s / segments) * Math.PI * 2 + time * (1.2 - t * 0.6) + t * 8;
          // Subtle organic wave distortion in the vortex
          const wobble = Math.sin(theta * 3 + time * 2 + t * 12) * (radius * 0.035);
          const r = radius + wobble;

          const px = cx + Math.cos(theta) * r;
          const py = cy + y + Math.sin(theta) * r * tilt;

          if (s === 0) ctx.moveTo(px, py);
          else ctx.lineTo(px, py);
        }
        ctx.strokeStyle = `rgba(255, 255, 255, ${alpha})`;
        ctx.lineWidth = 0.85;
        ctx.stroke();
      }

      // 2. Draw longitudinal twisting Meridian strands
      const NUM_STRANDS = 36;
      for (let s = 0; s < NUM_STRANDS; s++) {
        const baseAngle = (s / NUM_STRANDS) * Math.PI * 2;
        ctx.beginPath();
        const steps = 50;
        for (let j = 0; j <= steps; j++) {
          const t = j / steps;
          const { y, radius } = getFunnelProfile(t);
          const theta = baseAngle + t * 7.5 - time * 1.4;
          const px = cx + Math.cos(theta) * radius;
          const py = cy + y + Math.sin(theta) * radius * tilt;
          if (j === 0) ctx.moveTo(px, py);
          else ctx.lineTo(px, py);
        }
        ctx.strokeStyle = 'rgba(255, 255, 255, 0.07)';
        ctx.lineWidth = 0.75;
        ctx.stroke();
      }

      // 3. Draw spiraling luminous particles
      for (const p of particles) {
        p.t += p.speedT;
        if (p.t > 1) p.t -= 1;
        p.angle += p.speedAngle * (1.4 - p.t * 0.6);

        const { y, radius } = getFunnelProfile(p.t);
        const px = cx + Math.cos(p.angle) * radius;
        const py = cy + y + Math.sin(p.angle) * radius * tilt;

        // Depth cue: particles in front (sin > 0) are brighter
        const depthBoost = (Math.sin(p.angle) + 1) * 0.5;
        const finalAlpha = p.alpha * (0.35 + depthBoost * 0.65) * Math.sin(p.t * Math.PI);

        ctx.fillStyle = `rgba(255, 255, 255, ${finalAlpha})`;
        ctx.fillRect(px, py, p.size, p.size);
      }

      // 4. Central soft dark vignette behind hero text for crisp readability
      const textMask = ctx.createRadialGradient(cx, cy - height * 0.04, 10, cx, cy - height * 0.04, width * 0.32);
      textMask.addColorStop(0, 'rgba(0, 0, 0, 0.58)');
      textMask.addColorStop(0.6, 'rgba(0, 0, 0, 0.22)');
      textMask.addColorStop(1, 'rgba(0, 0, 0, 0)');
      ctx.fillStyle = textMask;
      ctx.fillRect(0, 0, width, height);

      animationFrameId = requestAnimationFrame(render);
    };

    render();

    return () => {
      cancelAnimationFrame(animationFrameId);
      window.removeEventListener('resize', handleResize);
    };
  }, [isDark]);

  if (!isDark) return null;

  return (
    <div className={`pointer-events-none absolute inset-0 overflow-hidden ${className}`}>
      <canvas ref={canvasRef} className="h-full w-full block" />
      {/* Bottom fade into Scene 2 */}
      <div className="
        pointer-events-none absolute inset-x-0 bottom-0 h-32
        bg-gradient-to-t from-black via-black/70 to-transparent
      " />
    </div>
  );
}
```

### 2. Scene 2: Ambient Vertical Light Streaks & Grid (`src/components/ui/Backdrop.jsx`)

Update `src/components/ui/Backdrop.jsx` so that in Dark Mode it renders the subtle vertical falling/rising light streaks, floating micro-dust particles, and architectural hairline borders with `+` crosshairs:

```jsx
import React, { useEffect, useRef } from 'react';
import { useTheme } from '../../theme/ThemeProvider';

export default function Backdrop({ className = '' }) {
  const canvasRef = useRef(null);
  const { theme } = useTheme?.() || { theme: 'dark' };
  const isDark = theme === 'dark' || document.documentElement.classList.contains('dark');

  useEffect(() => {
    if (!isDark) return;
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');

    let animationFrameId;
    let width = (canvas.width = window.innerWidth);
    let height = (canvas.height = window.innerHeight);

    const handleResize = () => {
      width = canvas.width = window.innerWidth;
      height = canvas.height = window.innerHeight;
    };
    window.addEventListener('resize', handleResize);

    // Vertical light streaks (Qronos Scene 2 style)
    const NUM_STREAKS = 65;
    const streaks = Array.from({ length: NUM_STREAKS }, () => ({
      x: Math.random() * width,
      y: Math.random() * height,
      length: 45 + Math.random() * 140,
      speed: 0.35 + Math.random() * 0.95,
      opacity: 0.05 + Math.random() * 0.22,
      hasHeadDot: Math.random() > 0.55,
    }));

    // Subtle ambient micro-dots
    const NUM_DOTS = 50;
    const dots = Array.from({ length: NUM_DOTS }, () => ({
      x: Math.random() * width,
      y: Math.random() * height,
      vy: 0.1 + Math.random() * 0.25,
      size: 1 + Math.random() * 1.2,
      alpha: 0.08 + Math.random() * 0.25,
    }));

    const render = () => {
      ctx.clearRect(0, 0, width, height);

      // Draw vertical streaks
      for (const s of streaks) {
        s.y -= s.speed; //缓慢 upward drift (change to += for downward rain)
        if (s.y + s.length < 0) {
          s.y = height + Math.random() * 80;
          s.x = Math.random() * width;
        }

        const grad = ctx.createLinearGradient(s.x, s.y, s.x, s.y + s.length);
        grad.addColorStop(0, `rgba(255, 255, 255, ${s.opacity})`);
        grad.addColorStop(0.3, `rgba(255, 255, 255, ${s.opacity * 0.6})`);
        grad.addColorStop(1, 'rgba(255, 255, 255, 0)');

        ctx.strokeStyle = grad;
        ctx.lineWidth = 1;
        ctx.beginPath();
        ctx.moveTo(s.x, s.y);
        ctx.lineTo(s.x, s.y + s.length);
        ctx.stroke();

        if (s.hasHeadDot) {
          ctx.fillStyle = `rgba(255, 255, 255, ${Math.min(0.45, s.opacity * 1.8)})`;
          ctx.fillRect(s.x - 0.5, s.y - 0.5, 2, 2);
        }
      }

      // Draw floating micro-dots
      for (const d of dots) {
        d.y -= d.vy;
        if (d.y < 0) {
          d.y = height;
          d.x = Math.random() * width;
        }
        ctx.fillStyle = `rgba(255, 255, 255, ${d.alpha})`;
        ctx.fillRect(d.x, d.y, d.size, d.size);
      }

      animationFrameId = requestAnimationFrame(render);
    };

    render();

    return () => {
      cancelAnimationFrame(animationFrameId);
      window.removeEventListener('resize', handleResize);
    };
  }, [isDark]);

  if (!isDark) {
    // Keep your existing light-mode backdrop intact
    return <div className={`pointer-events-none fixed inset-0 -z-10 ${className}`} />;
  }

  return (
    <div className={`pointer-events-none fixed inset-0 -z-10 bg-[#000000] overflow-hidden ${className}`}>
      {/* Vertical Light Streaks Canvas */}
      <canvas ref={canvasRef} className="h-full w-full block opacity-90" />

      {/* Architectural Outer Frame Rail Lines & Corner '+' Crosshairs */}
      <div className="pointer-events-none fixed inset-y-0 left-6 md:left-12 w-px bg-white/[0.08]" />
      <div className="pointer-events-none fixed inset-y-0 right-6 md:right-12 w-px bg-white/[0.08]" />
      <div className="pointer-events-none fixed inset-x-0 top-16 h-px bg-white/[0.08]" />
    </div>
  );
}
```

```text
=== CampusLink-Demo2/src/theme/ThemeProvider.jsx ===
import { createContext, useContext, useState, useCallback } from 'react';
import { flushSync } from 'react-dom';
const Ctx = createContext(null);
const KEY = 'cl-theme';
export function ThemeProvider({ children }) {
  const [theme, setThemeState] = useState(() => document.documentElement.dataset.theme || 'day');
  const apply = useCallback((t) => {
    document.documentElement.dataset.theme = t;
    try { localStorage.setItem(KEY, t); } catch { /* ignore */ }
    setThemeState(t);
  }, []);
  const toggle = useCallback(async (e) => {
    const next = theme === 'night' ? 'day' : 'night';
    const reduce = typeof matchMedia !== 'undefined' && matchMedia('(prefers-reduced-motion: reduce)').matches;
    if (!document.startViewTransition || reduce || !e) { apply(next); return; }
    const x = e.clientX, y = e.clientY;
    const r = Math.hypot(Math.max(x, innerWidth - x), Math.max(y, innerHeight - y));
    const vt = document.startViewTransition(() => { flushSync(() => apply(next)); });
    await vt.ready;
    document.documentElement.animate(
      { clipPath: [`circle(0px at ${x}px ${y}px)`, `circle(${r}px at ${x}px ${y}px)`] },
      { duration: 650, easing: 'ease-in-out', pseudoElement: '::view-transition-new(root)' }
    );
  }, [theme, apply]);
  return <Ctx.Provider value={{ theme, toggle }}>{children}</Ctx.Provider>;
}
export const useTheme = () => useContext(Ctx);

import React from 'react';
import { motion } from 'framer-motion';
import { cn } from '../../lib/utils';

export function Card({
  children,
  className,
  hover = false,
  object = false,
  onClick,
  ...props
}) {
  const Component = hover ? motion.div : 'div';
  const hoverProps = hover
    ? {
        whileHover: { y: -2, transition: { duration: 0.15 } },
      }
    : {};

  return (
    <Component
      onClick={onClick}
      className={cn(
        'relative overflow-hidden transition-all',
        object
          ? 'glass-strong rounded-object p-5 shadow-object'
          : 'bg-surface rounded-object border border-line p-5 shadow-subtle',
        hover && 'cursor-pointer hover:border-ink/40',
        className
      )}
      {...hoverProps}
      {...props}
    >
      {children}
    </Component>
  );
}
```

```text
export function Thread({ colors, radius = 0.045 }) {
  const geo = useMemo(() => {
    const g = new THREE.TubeGeometry(threadCurve, 300, radius, 8, false);
    const n = g.attributes.position.count;
    const arr = new Float32Array(n * 3);
    const a = new THREE.Color(colors.a);
    const b = new THREE.Color(colors.b);
    const tmp = new THREE.Color();
    for (let i = 0; i < n; i++) {
      tmp.copy(a).lerp(b, i / n);
      arr.set([tmp.r, tmp.g, tmp.b], i * 3);
    }
    g.setAttribute('color', new THREE.BufferAttribute(arr, 3));
    return g;
  }, [radius, colors.a, colors.b]);
  useEffect(() => () => geo.dispose(), [geo]);
  return (
    <mesh geometry={geo}>
      <meshBasicMaterial vertexColors toneMapped={false} />
    </mesh>
  );
}

export function Dust({ count = 320, color, additive }) {
  const ref = useRef();
  const positions = useMemo(() => {
    const a = new Float32Array(count * 3);
    for (let i = 0; i < count; i++) {
      a[i * 3] = (Math.random() - 0.5) * 18;
      a[i * 3 + 1] = (Math.random() - 0.5) * 9;
      a[i * 3 + 2] = -Math.random() * 12;
    }
    return a;
  }, [count]);
  useFrame((_, dt) => {
    if (ref.current) ref.current.rotation.y += dt * 0.01;
  });
  return (
    <Points ref={ref} positions={positions} stride={3} frustumCulled={false}>
      <PointMaterial
        transparent
        color={color}
        size={0.035}
        sizeAttenuation
        depthWrite={false}
        blending={additive ? THREE.AdditiveBlending : THREE.NormalBlending}
      />
    </Points>
  );
}

function PointerParallax({ children, amount = 0.12 }) {
  const ref = useRef();
  useFrame(({ pointer }, dt) => {
    if (!ref.current) return;
    ref.current.rotation.y = THREE.MathUtils.damp(ref.current.rotation.y, pointer.x * amount, 4, dt);
    ref.current.rotation.x = THREE.MathUtils.damp(ref.current.rotation.x, -pointer.y * amount * 0.6, 4, dt);
  });
  return <group ref={ref}>{children}</group>;
}

const _p = new THREE.Vector3();
const _l = new THREE.Vector3();
const _t = new THREE.Vector3();
const _off = new THREE.Vector3(0, 0.35, 1.4);

export function CameraRig({ progress }) {
  const { camera } = useThree();
  useFrame((_, dt) => {
    const val = typeof progress?.get === 'function' ? progress.get() : (progress || 0);
    const t = THREE.MathUtils.clamp(val, 0, 0.97);
    threadCurve.getPointAt(t, _p);
    threadCurve.getPointAt(Math.min(t + 0.03, 1), _l);
    _t.copy(_p).add(_off);
    camera.position.lerp(_t, 1 - Math.exp(-6 * dt));
    camera.lookAt(_l);
  });
  return null;
}
```

### 1. `src/three/HeroCanvas.jsx` (Scene 1: Qronos Vortex Funnel in Dark Mode)

In `'day'` mode, this keeps your original Three.js `HeroScene` thread. As soon as you switch to `'night'` mode, it renders the **Qronos Swirling 3D Vortex Funnel** and pauses automatically when scrolled out of view (`visible={inView}`).

```jsx
import React, { useEffect, useRef } from 'react';
import { SceneCanvas, HeroScene } from './Scenes.jsx';
import { useTheme } from '../theme/ThemeProvider.jsx';

export default function HeroCanvas({ visible = true }) {
  const { theme } = useTheme();
  const isNight = theme === 'night';
  const canvasRef = useRef(null);

  useEffect(() => {
    if (!isNight || !visible) return;
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');

    let animationFrameId;
    let width = (canvas.width = canvas.parentElement?.clientWidth || window.innerWidth);
    let height = (canvas.height = canvas.parentElement?.clientHeight || window.innerHeight);

    const handleResize = () => {
      if (!canvas.parentElement) return;
      width = canvas.width = canvas.parentElement.clientWidth;
      height = canvas.height = canvas.parentElement.clientHeight;
    };
    window.addEventListener('resize', handleResize);

    // Parametric funnel profile: wide top mouth -> narrow throat -> flared bottom disc
    const getFunnelProfile = (t) => {
      let radius;
      let y;
      if (t < 0.64) {
        const u = t / 0.64;
        y = -height * 0.56 + u * (height * 0.80);
        radius = width * 0.042 + Math.pow(1 - u, 2.15) * (width * 0.44);
      } else {
        const u = (t - 0.64) / 0.36;
        y = height * 0.24 + Math.pow(u, 0.72) * (height * 0.25);
        radius = width * 0.042 + Math.pow(u, 1.48) * (width * 0.70);
      }
      return { y, radius };
    };

    const NUM_PARTICLES = 850;
    const particles = Array.from({ length: NUM_PARTICLES }, () => ({
      t: Math.random(),
      angle: Math.random() * Math.PI * 2,
      speedT: 0.0005 + Math.random() * 0.0011,
      speedAngle: 0.01 + Math.random() * 0.016,
      size: Math.random() < 0.82 ? 1.1 : 1.8,
      alpha: 0.16 + Math.random() * 0.52,
    }));

    let time = 0;

    const render = () => {
      time += 0.011;
      ctx.fillStyle = '#000000';
      ctx.fillRect(0, 0, width, height);

      const cx = width * 0.5;
      const cy = height * 0.46;
      const tilt = 0.27;

      // 1. Swirling orbital spline rings
      const NUM_RINGS = 76;
      for (let i = 0; i < NUM_RINGS; i++) {
        const t = ((i / NUM_RINGS) + time * 0.022) % 1;
        const { y, radius } = getFunnelProfile(t);
        const edgeFade = Math.sin(t * Math.PI);
        const alpha = Math.max(0.03, edgeFade * 0.21);

        ctx.beginPath();
        const segments = 60;
        for (let s = 0; s <= segments; s++) {
          const theta = (s / segments) * Math.PI * 2 + time * (1.15 - t * 0.55) + t * 8;
          const wobble = Math.sin(theta * 3 + time * 1.8 + t * 11) * (radius * 0.032);
          const r = radius + wobble;

          const px = cx + Math.cos(theta) * r;
          const py = cy + y + Math.sin(theta) * r * tilt;

          if (s === 0) ctx.moveTo(px, py);
          else ctx.lineTo(px, py);
        }
        ctx.strokeStyle = `rgba(255, 255, 255, ${alpha})`;
        ctx.lineWidth = 0.85;
        ctx.stroke();
      }

      // 2. Twisting longitudinal meridian strands
      const NUM_STRANDS = 34;
      for (let s = 0; s < NUM_STRANDS; s++) {
        const baseAngle = (s / NUM_STRANDS) * Math.PI * 2;
        ctx.beginPath();
        const steps = 48;
        for (let j = 0; j <= steps; j++) {
          const t = j / steps;
          const { y, radius } = getFunnelProfile(t);
          const theta = baseAngle + t * 7.2 - time * 1.3;
          const px = cx + Math.cos(theta) * radius;
          const py = cy + y + Math.sin(theta) * radius * tilt;
          if (j === 0) ctx.moveTo(px, py);
          else ctx.lineTo(px, py);
        }
        ctx.strokeStyle = 'rgba(255, 255, 255, 0.065)';
        ctx.lineWidth = 0.75;
        ctx.stroke();
      }

      // 3. Flowing white/silver vortex particles
      for (const p of particles) {
        p.t += p.speedT;
        if (p.t > 1) p.t -= 1;
        p.angle += p.speedAngle * (1.35 - p.t * 0.55);

        const { y, radius } = getFunnelProfile(p.t);
        const px = cx + Math.cos(p.angle) * radius;
        const py = cy + y + Math.sin(p.angle) * radius * tilt;

        const depthBoost = (Math.sin(p.angle) + 1) * 0.5;
        const finalAlpha = p.alpha * (0.35 + depthBoost * 0.65) * Math.sin(p.t * Math.PI);

        ctx.fillStyle = `rgba(255, 255, 255, ${finalAlpha})`;
        ctx.fillRect(px, py, p.size, p.size);
      }

      // 4. Readability vignette behind Hero copy & candidate slips
      const vignette = ctx.createRadialGradient(
        cx - width * 0.12,
        cy - height * 0.04,
        20,
        cx,
        cy,
        width * 0.45
      );
      vignette.addColorStop(0, 'rgba(0, 0, 0, 0.68)');
      vignette.addColorStop(0.55, 'rgba(0, 0, 0, 0.32)');
      vignette.addColorStop(1, 'rgba(0, 0, 0, 0)');
      ctx.fillStyle = vignette;
      ctx.fillRect(0, 0, width, height);

      animationFrameId = requestAnimationFrame(render);
    };

    render();

    return () => {
      cancelAnimationFrame(animationFrameId);
      window.removeEventListener('resize', handleResize);
    };
  }, [isNight, visible]);

  if (!isNight) {
    return (
      <SceneCanvas visible={visible} camera={{ position: [0, 0, 8], fov: 50 }}>
        <HeroScene />
      </SceneCanvas>
    );
  }

  return (
    <div className="pointer-events-none absolute inset-0 overflow-hidden bg-black">
      <canvas ref={canvasRef} className="h-full w-full block" />
      <div className="
        pointer-events-none absolute inset-x-0 bottom-0 h-28
        bg-gradient-to-t from-black via-black/65 to-transparent
      " />
    </div>
  );
}
```

### 2. `src/components/ui/Backdrop.jsx` (Scene 2: Vertical Light Streaks & Grid Rails)

Note that your codebase uses a **named export** (`export function Backdrop()`). In `'day'` mode it renders your three drifting pastel blobs; in `'night'` mode it renders **Scene 2** (vertical light trails + micro-dust + left/right architectural frame lines) across both the lower Landing Page and every Student/Admin/Recruiter dashboard in `AppShell.jsx`.

```jsx
import React, { useEffect, useRef } from 'react';
import { useTheme } from '../../theme/ThemeProvider.jsx';

export function Backdrop() {
  const { theme } = useTheme();
  const isNight = theme === 'night';
  const canvasRef = useRef(null);

  useEffect(() => {
    if (!isNight) return;
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');

    let animationFrameId;
    let width = (canvas.width = window.innerWidth);
    let height = (canvas.height = window.innerHeight);

    const handleResize = () => {
      width = canvas.width = window.innerWidth;
      height = canvas.height = window.innerHeight;
    };
    window.addEventListener('resize', handleResize);

    // Qronos Scene 2 vertical light streaks
    const NUM_STREAKS = 64;
    const streaks = Array.from({ length: NUM_STREAKS }, () => ({
      x: Math.random() * width,
      y: Math.random() * height,
      length: 50 + Math.random() * 145,
      speed: 0.3 + Math.random() * 0.85,
      opacity: 0.05 + Math.random() * 0.20,
      hasHeadDot: Math.random() > 0.55,
    }));

    // Ambient floating micro-particles
    const NUM_DOTS = 48;
    const dots = Array.from({ length: NUM_DOTS }, () => ({
      x: Math.random() * width,
      y: Math.random() * height,
      vy: 0.08 + Math.random() * 0.22,
      size: 1 + Math.random() * 1.2,
      alpha: 0.07 + Math.random() * 0.22,
    }));

    const render = () => {
      ctx.clearRect(0, 0, width, height);

      for (const s of streaks) {
        s.y -= s.speed;
        if (s.y + s.length < 0) {
          s.y = height + Math.random() * 90;
          s.x = Math.random() * width;
        }

        const grad = ctx.createLinearGradient(s.x, s.y, s.x, s.y + s.length);
        grad.addColorStop(0, `rgba(255, 255, 255, ${s.opacity})`);
        grad.addColorStop(0.35, `rgba(255, 255, 255, ${s.opacity * 0.55})`);
        grad.addColorStop(1, 'rgba(255, 255, 255, 0)');

        ctx.strokeStyle = grad;
        ctx.lineWidth = 1;
        ctx.beginPath();
        ctx.moveTo(s.x, s.y);
        ctx.lineTo(s.x, s.y + s.length);
        ctx.stroke();

        if (s.hasHeadDot) {
          ctx.fillStyle = `rgba(255, 255, 255, ${Math.min(0.42, s.opacity * 1.75)})`;
          ctx.fillRect(s.x - 0.5, s.y - 0.5, 2, 2);
        }
      }

      for (const d of dots) {
        d.y -= d.vy;
        if (d.y < 0) {
          d.y = height;
          d.x = Math.random() * width;
        }
        ctx.fillStyle = `rgba(255, 255, 255, ${d.alpha})`;
        ctx.fillRect(d.x, d.y, d.size, d.size);
      }

      animationFrameId = requestAnimationFrame(render);
    };

    render();

    return () => {
      cancelAnimationFrame(animationFrameId);
      window.removeEventListener('resize', handleResize);
    };
  }, [isNight]);

  if (!isNight) {
    return (
      <div className="backdrop" aria-hidden="true">
        <i />
        <i />
        <i />
      </div>
    );
  }

  return (
    <div
      className="pointer-events-none fixed inset-0 z-0 bg-[#000000] overflow-hidden"
      aria-hidden="true"
    >
      <canvas ref={canvasRef} className="h-full w-full block opacity-90" />
      {/* Architectural vertical framing lines */}
      <div className="pointer-events-none fixed inset-y-0 left-4 lg:left-10 w-px bg-white/[0.08]" />
      <div className="pointer-events-none fixed inset-y-0 right-4 lg:right-10 w-px bg-white/[0.08]" />
    </div>
  );
}
```

### 4. `src/components/ui/Card.jsx` & `src/index.css` (Obsidian Glass Cards over Scene 2)

Update `src/components/ui/Card.jsx` so your Student, Recruiter, and Admin cards sit cleanly above the Scene 2 vertical streaks with Qronos obsidian glass styling in `'night'` mode:

```jsx
import React from 'react';
import { motion } from 'framer-motion';
import { cn } from '../../lib/utils';

export function Card({
  children,
  className,
  hover = false,
  object = false,
  onClick,
  ...props
}) {
  const Component = hover ? motion.div : 'div';
  const hoverProps = hover
    ? {
        whileHover: { y: -2, transition: { duration: 0.15 } },
      }
    : {};

  return (
    <Component
      onClick={onClick}
      className={cn(
        'relative overflow-hidden transition-all qronos-card',
        object
          ? 'glass-strong rounded-object p-5 shadow-object'
          : 'bg-surface rounded-object border border-line p-5 shadow-subtle',
        hover && 'cursor-pointer hover:border-ink/40',
        className
      )}
      {...hoverProps}
      {...props}
    >
      {children}
    </Component>
  );
}
```
