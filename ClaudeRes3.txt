Rules that stop glass looking like an AI template
1. Glass needs something colorful behind it. That is why the backdrop exists. Blur over a flat color just looks dirty.
2. At most six glass layers visible at once. No glass on table rows or list items. Use one large PageSheet per page, plus header, sidebar and floating objects.
3. Never more than two glass layers deep (PageSheet, then an object).
4. The edge is what reads as glass: a 1px light border plus an inner top highlight, not just blur.
5. Gradients are allowed in exactly three places: the backdrop, the pointer specular highlight on tilt objects, and 3D vertex colors. Everywhere else the old ban stands.
6. Radii: controls 6px, objects 10px, PageSheet 16px, stamps 3px.
7. Text sits on glass with at least 62% alpha in Day, and the contrast script covers the worst case.

18. The interaction layer (the masala)
Eight things, chosen because each one is visible within ten seconds and none of them hides information:
1. Cinematic intro: once per session, about 1.4s, skippable. The thread draws, the wordmark settles, the curtain lifts.
2. Smooth scroll on the landing page (Lenis).
3. Day/Night switch with a circular reveal from the toggle, using the View Transitions API where available (Chromium and Safari). Elsewhere it switches instantly.
4. Glass tilt with a moving specular highlight on floating objects.
5. Magnetic primary buttons, on three buttons only.
6. The register ticker: a slow mono marquee of simulated events under the hero. Pauses on hover.
7. Portal page-turn: a curtain wipe when you switch role, so the most-used judge action feels good.
8. Hero parallax: three depth layers (backdrop, canvas, slips) respond to the pointer.
The Judge's Guide grows from 6 to 8 steps: 7 switch Day/Night, 8 scroll the landing pipeline.
Skipped on purpose: sound, cursor blobs, confetti, bloom, an auto-playing demo, 3D inside data screens, and model files.

// Generates src/styles/themes.generated.css from src/theme/tokens.js. Runs before dev and build.
import { writeFileSync, mkdirSync } from 'node:fs';
import { tokens } from '../src/theme/tokens.js';
const kebab = (s) => s.replace(/[A-Z]/g, (m) => '-' + m.toLowerCase());
const isHex = (v) => typeof v === 'string' && /^#[0-9a-fA-F]{6}$/.test(v);
const triplet = (h) => [1, 3, 5].map((i) => parseInt(h.slice(i, i + 2), 16)).join(' ');
const block = (name, t) => {
  const lines = [];
  for (const [k, v] of Object.entries(t)) {
    if (k === 'checkBackdrops') continue;
    if (k.startsWith('three')) lines.push(`  --${kebab(k)}: ${v};`);
    else if (isHex(v)) lines.push(`  --c-${kebab(k)}: ${triplet(v)};`);
    else lines.push(`  --${kebab(k)}: ${v};`);
  }
  return `:root[data-theme='${name}'] {\n  color-scheme: ${name === 'night' ? 'dark' : 'light'};\n${lines.join('\n')}\n}\n`;
};
mkdirSync('src/styles', { recursive: true });
writeFileSync('src/styles/themes.generated.css', `/* GENERATED from src/theme/tokens.js. Do not edit. */\n${block('day', tokens.day)}\n${block('night', tokens.night)}`);
console.log('themes.generated.css written');

// WCAG contrast check for both themes, including worst-case backdrops under glass. Exit code 1 on failure.
import { tokens } from '../src/theme/tokens.js';
const hex = (h) => [1, 3, 5].map((i) => parseInt(h.slice(i, i + 2), 16));
const lin = (v) => { v /= 255; return v <= 0.03928 ? v / 12.92 : Math.pow((v + 0.055) / 1.055, 2.4); };
const lum = ([r, g, b]) => 0.2126 * lin(r) + 0.7152 * lin(g) + 0.0722 * lin(b);
const ratio = (a, b) => { const [x, y] = [lum(a), lum(b)].sort((p, q) => q - p); return (x + 0.05) / (y + 0.05); };
const over = (fg, alpha, bg) => fg.map((c, i) => Math.round(c * alpha + bg[i] * (1 - alpha)));
let fail = 0, lowest = 99;
for (const [name, t] of Object.entries(tokens)) {
  console.log(`\n== ${name.toUpperCase()} ==`);
  const check = (label, a, b, min = 4.5) => {
    const r = ratio(a, b); lowest = Math.min(lowest, r);
    if (r < min) { fail++; console.log(`FAIL ${r.toFixed(2)} (min ${min}) ${label}`); }
  };
  for (const bd of t.checkBackdrops) {
    const glass = over(hex(t.glassBase), parseFloat(t.glassAlpha), hex(bd));
    const mark = over(hex(t.accent), parseFloat(t.markAlpha), glass);
    for (const k of ['fg', 'fgSoft', 'jade', 'brick', 'lagoon']) check(`${k} on glass over ${bd}`, hex(t[k]), glass);
    check(`fg on highlighter mark over ${bd}`, hex(t.fg), mark);
    if (name === 'night') check(`accent text on glass over ${bd}`, hex(t.accent), glass);
  }
  check('accentDeep on accentTint', hex(t.accentDeep), hex(t.accentTint));
  check('jade on jadeTint', hex(t.jade), hex(t.jadeTint));
  check('brick on brickTint', hex(t.brick), hex(t.brickTint));
  check('btnFg on btn', hex(t.btnFg), hex(t.btn));
  check('btnHoverFg on btnHover', hex(t.btnHoverFg), hex(t.btnHover));
  check('sidebarFg on sidebar', hex(t.sidebarFg), hex(t.sidebar));
  check('darkSectionFg on darkSection', hex(t.darkSectionFg), hex(t.darkSection));
  check('fg on surface', hex(t.fg), hex(t.surface));
  check('fg on accent (text on solid accent)', name === 'day' ? hex(t.fg) : hex(t.btnFg), hex(t.accent));
}
console.log(fail ? `\n${fail} contrast check(s) FAILED` : `\nALL CONTRAST CHECKS PASSED (lowest ratio ${lowest.toFixed(2)})`);

import { useRef, useCallback } from 'react';
const reduced = () => typeof matchMedia !== 'undefined' && matchMedia('(prefers-reduced-motion: reduce)').matches;
export function useTilt({ max = 6 } = {}) {
  const ref = useRef(null);
  const onPointerMove = useCallback((e) => {
    const el = ref.current; if (!el || reduced()) return;
    const r = el.getBoundingClientRect();
    const px = (e.clientX - r.left) / r.width, py = (e.clientY - r.top) / r.height;
    el.style.setProperty('--rx', `${((0.5 - py) * max * 2).toFixed(2)}deg`);
    el.style.setProperty('--ry', `${((px - 0.5) * max * 2).toFixed(2)}deg`);
    el.style.setProperty('--gx', `${(px * 100).toFixed(1)}%`);
    el.style.setProperty('--gy', `${(py * 100).toFixed(1)}%`);
  }, [max]);
  const onPointerLeave = useCallback(() => {
    const el = ref.current; if (!el) return;
    el.style.setProperty('--rx', '0deg'); el.style.setProperty('--ry', '0deg');
  }, []);
  return { ref, onPointerMove, onPointerLeave };
}
export function useMagnetic({ strength = 0.25, radius = 90 } = {}) {
  const ref = useRef(null);
  const onPointerMove = useCallback((e) => {
    const el = ref.current; if (!el || reduced()) return;
    const r = el.getBoundingClientRect();
    const dx = e.clientX - (r.left + r.width / 2), dy = e.clientY - (r.top + r.height / 2);
    if (Math.hypot(dx, dy) > Math.max(r.width, r.height) / 2 + radius) return;
    el.style.transform = `translate(${(dx * strength).toFixed(1)}px, ${(dy * strength).toFixed(1)}px)`;
  }, [strength, radius]);
  const onPointerLeave = useCallback(() => { const el = ref.current; if (el) el.style.transform = ''; }, []);
  return { ref, onPointerMove, onPointerLeave };
}

import { useEffect, useState } from 'react';
export function useInView(ref, margin = '200px') {
  const [visible, setVisible] = useState(false);
  useEffect(() => {
    const el = ref.current; if (!el) return;
    const io = new IntersectionObserver(([e]) => setVisible(e.isIntersecting), { rootMargin: margin });
    io.observe(el);
    return () => io.disconnect();
  }, [ref, margin]);
  return visible;
}

export function Thread({ colors, radius = 0.045 }) {
  const geo = useMemo(() => {
    const g = new THREE.TubeGeometry(threadCurve, 300, radius, 8, false);
    const n = g.attributes.position.count, arr = new Float32Array(n * 3);
    const a = new THREE.Color(colors.a), b = new THREE.Color(colors.b), tmp = new THREE.Color();
    for (let i = 0; i < n; i++) { tmp.copy(a).lerp(b, i / n); arr.set([tmp.r, tmp.g, tmp.b], i * 3); }
    g.setAttribute('color', new THREE.BufferAttribute(arr, 3));
    return g;
  }, [radius, colors.a, colors.b]);
  useEffect(() => () => geo.dispose(), [geo]);
  return <mesh geometry={geo}><meshBasicMaterial vertexColors toneMapped={false} /></mesh>;
}

export function Dust({ count = 320, color, additive }) {
  const ref = useRef();
  const positions = useMemo(() => {
    const a = new Float32Array(count * 3);
    for (let i = 0; i < count; i++) { a[i * 3] = (Math.random() - 0.5) * 18; a[i * 3 + 1] = (Math.random() - 0.5) * 9; a[i * 3 + 2] = -Math.random() * 12; }
    return a;
  }, [count]);
  useFrame((_, dt) => { if (ref.current) ref.current.rotation.y += dt * 0.01; });
  return (
    <Points ref={ref} positions={positions} stride={3} frustumCulled={false}>
      <PointMaterial transparent color={color} size={0.035} sizeAttenuation depthWrite={false} blending={additive ? THREE.AdditiveBlending : THREE.NormalBlending} />
    </Points>
  );
}

function PointerParallax({ children, amount = 0.12 }) {
  const ref = useRef();
  useFrame(({ pointer }, dt) => {
    if (!ref.current) return;
    ref.current.rotation.y = THREE.MathUtils.damp(ref.current.rotation.y, pointer.x * amount, 4, dt);
    ref.current.rotation.x = THREE.MathUtils.damp(ref.current.rotation.x, -pointer.y * amount * 0.6, 4, dt);
  });
  return <group ref={ref}>{children}</group>;
}

const _p = new THREE.Vector3(), _l = new THREE.Vector3(), _t = new THREE.Vector3(), _off = new THREE.Vector3(0, 0.35, 1.4);
export function CameraRig({ progress }) {
  const { camera } = useThree();
  useFrame((_, dt) => {
    const t = THREE.MathUtils.clamp(progress.get(), 0, 0.97);
    threadCurve.getPointAt(t, _p); threadCurve.getPointAt(Math.min(t + 0.03, 1), _l);
    _t.copy(_p).add(_off);
    camera.position.lerp(_t, 1 - Math.exp(-6 * dt));
    camera.lookAt(_l);
  });
  return null;
}
