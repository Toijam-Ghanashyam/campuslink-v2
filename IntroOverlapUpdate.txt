// Inset a convex 2D polygon by `insetDist` pixels and build a rounded THREE.Shape
function createRoundedPolygonShape(pts, insetDist, cornerRadius) {
  const n = pts.length;
  // Compute centroid
  let cx = 0, cy = 0;
  for (const p of pts) {
    cx += p.x;
    cy += p.y;
  }
  cx /= n;
  cy /= n;

  // Inset vertices toward centroid
  const insetPts = pts.map((p) => {
    const dx = p.x - cx;
    const dy = p.y - cy;
    const len = Math.hypot(dx, dy) || 1;
    const shrink = Math.min(insetDist, len * 0.35);
    return {
      x: p.x - (dx / len) * shrink,
      y: p.y - (dy / len) * shrink,
    };
  });

  const shape = new THREE.Shape();
  for (let i = 0; i < n; i++) {
    const prev = insetPts[(i - 1 + n) % n];
    const curr = insetPts[i];
    const next = insetPts[(i + 1) % n];

    const v1x = prev.x - curr.x;
    const v1y = prev.y - curr.y;
    const v2x = next.x - curr.x;
    const v2y = next.y - curr.y;

    const l1 = Math.hypot(v1x, v1y) || 1;
    const l2 = Math.hypot(v2x, v2y) || 1;
    const r = Math.min(cornerRadius, l1 * 0.32, l2 * 0.32);

    const pStart = { x: curr.x + (v1x / l1) * r, y: curr.y + (v1y / l1) * r };
    const pEnd = { x: curr.x + (v2x / l2) * r, y: curr.y + (v2y / l2) * r };

    if (i === 0) {
      shape.moveTo(pStart.x, pStart.y);
    } else {
      shape.lineTo(pStart.x, pStart.y);
    }
    shape.quadraticCurveTo(curr.x, curr.y, pEnd.x, pEnd.y);
  }
  shape.closePath();
  return { shape, centroid: { x: cx, y: cy } };
}

        avy: (Math.random() - 0.5) * 5.2,
        avz: (Math.random() - 0.5) * 3.4,
        delay: idx === 0 ? 0 : 0.025 * idx,
      };
    });

    // 5. Pointer Raycasting & Interaction State
    const raycaster = new THREE.Raycaster();
    const pointerNDC = new THREE.Vector2(0, 0);
    let pointerActive = false;
    let hoveredIdx = -1;
    let isShattering = false;
    let shatterElapsed = 0;
    let introElapsed = 0;

    const handlePointerMove = (e) => {
      pointerActive = true;
      pointerNDC.x = (e.clientX / width) * 2 - 1;
      pointerNDC.y = -(e.clientY / height) * 2 + 1;

      uPointerWorld.value.set(
        (e.clientX - width * 0.5),
        -(e.clientY - height * 0.5),
        20
      );
    };

    const handlePointerLeave = () => {
      pointerActive = false;
      hoveredIdx = -1;
    };

    // Trigger the 3D shatter explosion
    shatterTriggerRef.current = (clickX = width * 0.5, clickY = height * 0.5) => {
      if (isShattering) return;
      isShattering = true;
      setShattering(true);

      const impactX = clickX - width * 0.5;
      const impactY = -(clickY - height * 0.5);

      shards.forEach((s) => {
        const dx = s.centroid.x - impactX;
        const dy = s.centroid.y - impactY;
        const angle = Math.atan2(dy, dx);
        const dist = Math.hypot(dx, dy);
        s.vx = Math.cos(angle) * (320 + dist * 0.95) * S + (Math.random() - 0.5) * 140 * S;
        s.vy = Math.sin(angle) * (300 + dist * 0.85) * S + 160 * S;
        s.vz = (580 + Math.random() * 380) * S;
      });
    };

    const handleKeyDown = (e) => {
      if (e.key === 'Enter' || e.key === ' ' || e.key === 'Escape') {
        shatterTriggerRef.current(width * 0.5, height * 0.5);
      }
    };

    const handleResize = () => {
      width = window.innerWidth;
      height = window.innerHeight;
      camera.aspect = width / height;
      camera.position.z = calcCameraZ(height);
      camera.updateProjectionMatrix();
      renderer.setSize(width, height);
    };

    window.addEventListener('pointermove', handlePointerMove);
    window.addEventListener('pointerleave', handlePointerLeave);
    window.addEventListener('keydown', handleKeyDown);
    window.addEventListener('resize', handleResize);

    // 6. Animation Loop
    const clock = new THREE.Clock();
    let frameId;

    const animate = () => {
      const dt = Math.min(clock.getDelta(), 0.05);
      introElapsed += dt;

      // Global block tilt toward pointer
      const targetRotY = pointerActive && !isShattering
        ? THREE.MathUtils.degToRad(pointerNDC.x * 7.0 - 1.0)
        : 0;
      const targetRotX = pointerActive && !isShattering
        ? THREE.MathUtils.degToRad(-pointerNDC.y * 5.5 + 0.8)
        : 0;

      rootGroup.rotation.y = THREE.MathUtils.damp(rootGroup.rotation.y, targetRotY, 6, dt);
      rootGroup.rotation.x = THREE.MathUtils.damp(rootGroup.rotation.x, targetRotX, 6, dt);

      uGlowStrength.value = THREE.MathUtils.damp(
        uGlowStrength.value,
        pointerActive && !isShattering ? 1.0 : 0.0,
        6,
        dt
      );

      if (!isShattering) {
        // Raycast to find hovered shard
        if (pointerActive && introElapsed > 0.45) {
          raycaster.setFromCamera(pointerNDC, camera);
          const hits = raycaster.intersectObjects(shards.map((s) => s.glassMesh));
          hoveredIdx = hits.length > 0 ? hits[0].object.userData.shardIndex : -1;
        } else {
          hoveredIdx = -1;
        }

        // Spring each shard into place + hover lift & tip
        shards.forEach((s) => {
          const isHovered = s.idx === hoveredIdx;
          const targetZ = isHovered ? 50 * S : 0;

          // Spring physics (k=170, c=19)
          const force = 170 * (targetZ - s.zPos) - 19 * s.zVel;
          s.zVel += force * dt;
          s.zPos += s.zVel * dt;
          s.group.position.z = s.zPos;

          // Tip shard away from pointer on hover
          let targetRx = 0;
          let targetRy = 0;
          if (isHovered) {
            const dx = (uPointerWorld.value.x - s.centroid.x) / (220 * S);
            const dy = (uPointerWorld.value.y - s.centroid.y) / (220 * S);
            targetRy = THREE.MathUtils.clamp(dx * 0.18, -0.24, 0.24);
            targetRx = THREE.MathUtils.clamp(-dy * 0.18, -0.24, 0.24);
          }

          s.rx = THREE.MathUtils.damp(s.rx, targetRx, 10, dt);
          s.ry = THREE.MathUtils.damp(s.ry, targetRy, 10, dt);
          s.rz = THREE.MathUtils.damp(s.rz, 0, 10, dt);
          s.group.rotation.set(s.rx, s.ry, s.rz);

          // Boost iridescence and reflections on hover
          s.glassMat.iridescence = THREE.MathUtils.damp(
            s.glassMat.iridescence,
            isHovered ? 1.0 : 0.55,
            8,
            dt
          );
          s.glassMat.envMapIntensity = THREE.MathUtils.damp(
            s.glassMat.envMapIntensity,
            isHovered ? 1.95 : 1.05,
            8,
            dt
          );
        });
      } else {
        // Shatter Explosion Physics
        shatterElapsed += dt;
        shards.forEach((s) => {
          const t = Math.max(0, shatterElapsed - s.delay);
          if (t <= 0) return;

          s.vy -= 920 * S * dt; // Gravity
          s.group.position.x += s.vx * dt;
          s.group.position.y += s.vy * dt;
          s.group.position.z += s.vz * dt;

          s.group.rotation.x += s.avx * dt;
          s.group.rotation.y += s.avy * dt;
          s.group.rotation.z += s.avz * dt;
        });

        if (shatterElapsed >= 0.88) {
          setShow(false);
          return;
        }
      }

      renderer.render(scene, camera);
      frameId = requestAnimationFrame(animate);
    };

    frameId = requestAnimationFrame(animate);

    return () => {
      cancelAnimationFrame(frameId);
      window.removeEventListener('pointermove', handlePointerMove);
      window.removeEventListener('pointerleave', handlePointerLeave);
      window.removeEventListener('keydown', handleKeyDown);
      window.removeEventListener('resize', handleResize);
      shards.forEach((s) => {
        s.extrudeGeo.dispose();
        s.faceGeo.dispose();
        s.glassMat.dispose();
      });
      masterTexture.dispose();
      envTex.dispose();
      pmrem.dispose();
      renderer.dispose();
    };
  }, [show]);

  if (!show) return null;

  return (
    <div
      onClick={(e) => handleShatter(e.clientX, e.clientY)}
      className={`fixed inset-0 z-[100] cursor-pointer select-none transition-colors duration-500 ${
        shattering ? 'bg-transparent pointer-events-none' : 'bg-[#050507]'
      }`}
      aria-label="CampusLink Intro Glass Monolith — Click anywhere to shatter and enter"
    >
      <div ref={mountRef} className="w-full h-full" />
    </div>
  );
}
How It Works Together on Your Landing Page:When you open or reload the Landing Page: IntroOverlay mounts immediately at z-[100] (bg-[#050507]). The 8 outer smoked-glass shards and the central core shard fly in and lock together into a thick, beveled obsidian monolith with CampusLink.register printed across the fracture lines.While moving your cursor: The whole glass slab tilts toward your pointer, a lavender subsurface glow illuminates the beveled cuts, and whichever shard is under your cursor springs toward the camera (+50px), tipping and refracting its slice of the CampusLink.register title.When you click anywhere (or press Enter): The #050507 background turns transparent immediately while the 9 thick glass shards explode outward and tumble past the camera—revealing your Qronos Dark Mode Vortex Landing Page spinning directly behind the flying shards!
